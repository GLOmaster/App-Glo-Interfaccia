// --- NUOVA FUNZIONE CONTATORE MINUTI CON DEBUG --- async function aggiornaMinuti() { if(!GITHUB_TOKEN) return; const url = `https://api.github.com/users/${GITHUB_USER}/settings/billing/actions`; const divMinuti = document.getElementById('minuti-residui'); try { divMinuti.innerText = "..."; divMinuti.style.color = "#38bdf8"; const res = await fetch(url, { headers: { 'Accept': 'application/vnd.github.v3+json', 'Authorization': `Bearer ${GITHUB_TOKEN}` } }); if(res.ok) { const data = await res.json(); const minutiUsati = data.total_minutes_used; const minutiRestanti = 2000 - minutiUsati; divMinuti.innerText = minutiRestanti; if(minutiRestanti < 500) divMinuti.style.color = "#fbbf24"; if(minutiRestanti < 100) divMinuti.style.color = "#ef4444"; } else { // SE FALLISCE, MOSTRA L'ERRORE ESATTO DI GITHUB const erroreEsatto = await res.text(); alert("Errore GitHub: " + res.status + "\nDettagli: " + erroreEsatto); divMinuti.innerText = "Errore API"; divMinuti.style.fontSize = "18px"; } } catch(e) { divMinuti.innerText = "Errore Rete"; divMinuti.style.fontSize = "18px"; } }
